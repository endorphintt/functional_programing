----Fibonacci

let fib n =
  let rec go a b i =
    if i = 0 then a else go b (a + b) (i - 1)
  in
  go 0 1 n


-----Sublists

let sublists (xs: 'a list) : 'a list list =
  let rec go xs =
    match xs with
    | [] -> [ [] ]
    | x :: tl ->
      let rest = go tl in
      rest @ List.map (fun ys -> x :: ys) rest
  in
  go xs

--- Rotate left

let rotate_left (k: int) (xs: 'a list) : 'a list =
  match xs with
  | [] -> []
  | _ ->
    let n = List.length xs in
    let k = ((k mod n) + n) mod n in
    let rec split i acc ys =
      if i = 0 then (List.rev acc, ys)
      else match ys with
        | [] -> (List.rev acc, [])
        | h :: tl -> split (i - 1) (h :: acc) tl
    in
    let a, b = split k [] xs in
    b @ a

--- Count occurrences

let count_occurrences (needle: string) (haystack: string list) : int =
  let rec go acc xs =
    match xs with
    | [] -> acc
    | h :: tl -> go (if h = needle then acc + 1 else acc) tl
  in
  go 0 haystack

let suffixes (xs: 'a list) : 'a list list =
  let rec go acc ys =
    match ys with
    | [] -> List.rev ([] :: acc)
    | _ :: tl -> go (ys :: acc) tl
  in
  go [] xs

let prefixes (xs: 'a list) : 'a list list =
  let rec go pref acc ys =
    match ys with
    | [] -> List.rev (List.rev pref :: acc)
    | h :: tl -> go (h :: pref) (List.rev pref :: acc) tl
  in
  go [] [] xs
